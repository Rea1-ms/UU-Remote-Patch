cmake_minimum_required(VERSION 3.24)
project(uu_adb_patch LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(nrd_probe tools/nrd-probe.cpp)
target_compile_options(nrd_probe PRIVATE /W4 /utf-8)
target_link_libraries(nrd_probe PRIVATE d3d11 dxgi)
add_executable(manager_probe tools/manager-probe.cpp)
target_compile_options(manager_probe PRIVATE /W4 /utf-8)
add_library(uu_core STATIC src/common.cpp src/frame.cpp src/input.cpp src/session.cpp src/texture.cpp)
target_include_directories(uu_core PUBLIC src third_party/json third_party/ffmpeg)
target_compile_options(uu_core PRIVATE /W4 /utf-8)
target_link_libraries(uu_core PUBLIC ws2_32 d3d11 dxgi)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(uu_patch SHARED src/proxy.cpp src/proxy.asm src/proxy.def src/hooks.cpp)
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/uu_patch.dir/${config}/src")
endforeach()
set_target_properties(uu_patch PROPERTIES OUTPUT_NAME version)
target_compile_options(uu_patch PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>)
target_link_libraries(uu_patch PRIVATE uu_core minhook bcrypt)
target_link_options(uu_patch PRIVATE /DEBUG /MAP)
target_compile_options(uu_patch PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/Zi>)
target_compile_options(uu_core PRIVATE /Zi)
add_executable(bridge_check tools/bridge-check.cpp)
target_compile_options(bridge_check PRIVATE /W4 /utf-8)
target_link_libraries(bridge_check PRIVATE uu_core)
enable_testing()
add_test(NAME protocol COMMAND bridge_check protocol)
add_test(NAME texture COMMAND bridge_check texture)
add_executable(patch_load tools/patch-load.cpp)
target_compile_options(patch_load PRIVATE /W4 /utf-8)
add_executable(proxy_check tools/proxy-check.cpp)
target_compile_options(proxy_check PRIVATE /W4 /utf-8)
add_test(NAME proxy COMMAND proxy_check $<TARGET_FILE:uu_patch>)
add_executable(management_check tools/management-check.cpp src/hooks.cpp)
target_compile_definitions(management_check PRIVATE UU_PATCH_CONTRACT_CHECK)
target_compile_options(management_check PRIVATE /W4 /utf-8)
target_link_libraries(management_check PRIVATE uu_core minhook bcrypt)
